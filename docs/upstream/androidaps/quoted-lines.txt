  103      override fun applyBasalConstraints(absoluteRate: Constraint<Double>, profile: Profile): Constraint<Double> {
  104          absoluteRate.setIfGreater(0.0, rh.gs(CoreUiStrings.limitingbasalratio, 0.0, rh.gs(CoreUiStrings.itmustbepositivevalue)), this)
  105          absoluteRate.setIfSmaller(hardLimits.maxBasal(), rh.gs(CoreUiStrings.limitingbasalratio, hardLimits.maxBasal(), rh.gs(ConstraintsStrings.hardlimit)), this)
  106          val pump = activePlugin.activePump
  107          // check for pump max
  108          if (pump.pumpDescription.tempBasalStyle == PumpDescription.ABSOLUTE) {
  109              // Concentration-adjusted max (the wrapper scales maxTempAbsolute by concentration); the raw
  110              // pumpType.tbrSettings().maxDose is in pump units (cU) and would mis-cap an IU rate under U200/diluted.
  111              val pumpLimit = pump.pumpDescription.maxTempAbsolute
  112              absoluteRate.setIfSmaller(pumpLimit, rh.gs(CoreUiStrings.limitingbasalratio, pumpLimit, rh.gs(CoreUiStrings.pumplimit)), this)
  113          }
  114  
  115          // do rounding — floor, not round-to-nearest, so a max-constrained rate is never rounded
  116          // back UP above the limit (and we never enact more basal than allowed). Effect is bounded by
  117          // one pump step; for already step-aligned values it is a no-op.
  118          if (pump.pumpDescription.tempBasalStyle == PumpDescription.ABSOLUTE) {
  119              absoluteRate.set(Round.floorTo(absoluteRate.value(), pump.pumpDescription.tempAbsoluteStep))
  120          }
  121          return absoluteRate
  122      }
